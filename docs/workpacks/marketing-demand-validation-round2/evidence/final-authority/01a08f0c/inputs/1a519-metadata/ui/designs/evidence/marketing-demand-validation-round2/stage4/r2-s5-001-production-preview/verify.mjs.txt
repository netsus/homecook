import {chromium,expect} from '@playwright/test';
import {readFile,writeFile} from 'node:fs/promises';
import {join,resolve} from 'node:path';
const out=resolve('.omx/artifacts/r2-s5-001/production-preview'),origin='http://127.0.0.1:3126';
const build=JSON.parse(await readFile(join(out,'build-result.json'),'utf8'));
const browser=await chromium.launch({headless:true});const checks=[],pageErrors=[],writes=[],external=[],headers=[];
const names={example:'사용 예시 먼저 보기',survey:'의견만 남기기 · 4문항',lead:'베타 오픈 알림 받기'};
const done={example:'사용 예시를 확인했어요',survey:'의견을 남겨주셔서 감사해요',lead:'베타 오픈 알림 신청을 접수했어요'};
try{for(const topic of ['recording','homeflow'])for(const activity of ['example','survey','lead']){
 const context=await browser.newContext({viewport:{width:390,height:844}});
 await context.route('**/*',route=>{const request=route.request(),url=new URL(request.url());if(url.origin!==origin){external.push(url.origin);return route.abort();}if(!['GET','HEAD','OPTIONS'].includes(request.method())){writes.push({method:request.method(),path:url.pathname});return route.abort();}return route.continue();});
 const page=await context.newPage();page.on('pageerror',error=>pageErrors.push(error.message));
 const response=await page.goto(origin+'/beta/r2/'+topic);expect(response.status()).toBe(200);const h=response.headers();headers.push({topic,activity,status:response.status(),cacheControl:h['cache-control'],referrerPolicy:h['referrer-policy'],setCookiePresent:Boolean(h['set-cookie'])});
 expect(h['cache-control']).toContain('private');expect(h['cache-control']).toContain('no-store');expect(h['referrer-policy']).toBe('no-referrer');expect(h['set-cookie']).toBeUndefined();
 await expect(page.getByText('로컬 미리보기 · 저장되지 않아요',{exact:true})).toBeVisible();
 await expect(page.getByText('아직 완료한 활동이 없어요',{exact:true})).toBeVisible();
 if(activity==='example')await page.screenshot({path:join(out,`${topic}-menu.png`),fullPage:true});
 await page.getByRole('button',{name:names[activity],exact:true}).click();
 await page.screenshot({path:join(out,`${topic}-${activity}.png`),fullPage:true,maskColor:'#111',mask:activity==='lead'?[page.getByRole('textbox',{name:'이메일',exact:true}),page.getByText(/^예시 이메일 /)]:[]});
 if(activity==='example'){for(let n=0;n<2;n++)await page.getByRole('button',{name:'다음 장면',exact:true}).click();await page.getByRole('button',{name:'예시 확인 완료',exact:true}).click();}
 if(activity==='survey'){for(let n=0;n<4;n++){await page.getByRole('radio').first().check();await page.getByRole('button',{name:n===3?'의견 보내기':'다음 문항',exact:true}).click();}}
 if(activity==='lead'){await page.getByRole('checkbox').check();await page.getByRole('button',{name:'베타 오픈 알림 신청하기',exact:true}).click();}
 await expect(page.getByRole('heading',{level:1,name:done[activity],exact:true})).toBeVisible();
 await page.getByRole('button',{name:'메뉴로 돌아가기',exact:true}).click();await expect(page.locator('nav[aria-label="원하는 활동 선택"]').getByText('완료',{exact:true})).toHaveCount(1);
 const storage=await page.evaluate(async()=>({local:localStorage.length,session:sessionStorage.length,idb:(await indexedDB.databases()).length}));expect(storage).toEqual({local:0,session:0,idb:0});
 await page.reload();await expect(page.getByText('아직 완료한 활동이 없어요',{exact:true})).toBeVisible();
 checks.push({topic,activity,result:'PASS',independentCompletion:true,memoryOnlyResetOnReload:true,storage});await context.close();
}expect(pageErrors).toEqual([]);expect(writes).toEqual([]);expect(external).toEqual([]);
 await writeFile(join(out,'headers.json'),JSON.stringify(headers,null,2));
 await writeFile(join(out,'verification.json'),JSON.stringify({result:'PASS',checkedAt:new Date().toISOString(),head:build.head,buildId:build.buildId,sourceTreeSha256:build.sourceTreeSha256,compiledTreeSha256:build.compiledTreeSha256,checks,pageErrors,writes,external,postCount:0,dbCalls:0,providerCalls:0,actualServerPage:true,fixturePageOverride:false,memoryPreview:true,publicStage5Approval:false},null,2));
 process.stdout.write(JSON.stringify({result:'PASS',checks:checks.length,head:build.head,buildId:build.buildId})+'\n');
}finally{await browser.close();}
