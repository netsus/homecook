function load(p) {
  let source;
  for (const candidate of [p, p + '.ts', p + '/index.ts']) {
    try { source = cp.execFileSync('git', ['show', sha + ':' + candidate], { cwd, encoding: 'utf8', stdio: ['ignore', 'pipe', 'ignore'] }); p = candidate; break; } catch {}
  }
  if (source === undefined) throw Error(p);
  if (cache.has(p)) return cache.get(p).exports;
  sources.push({ path: p, sha256: crypto.createHash('sha256').update(source).digest('hex') });
